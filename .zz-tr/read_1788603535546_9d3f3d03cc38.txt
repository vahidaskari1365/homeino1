    54→  const [imageBase64, setImageBase64] = useState<string | null>(null);

    57→  const [prompt, setPrompt] = useState("");

    65→  const [error, setError] = useState<string | null>(null);

    71→  const addToCart = useCart((s) => s.add);

    73→  const { toast } = useUi();

    78→  const rs = useRoomState();
    79→  const [, setLastIntent] = useState<AiIntent | null>(null);

    84→  const analyzeRoom = useCallback(async (image: string) => {

    91→    finally { setAnalyzing(false); }

    93→
    94→  const toggleCat = useCallback((slug: string) => setOpenCats((s) => { const n = new Set(s); n.has(slug) ? n.delete(slug) : n.add(slug); return n; }), []);
    95→  const toggleSubType = useCallback((slug: string, label: string) => setSelectedSubTypes((s) => { const cur = s[slug] || []; return { ...s, [slug]: cur.includes(label) ? cur.filter((x) => x !== label) : [...cur, label] }; }), []);

    97→  const selectStyle = useCallback((id: string) => { setStyle(id); if (id === "office") setOpenCats((s) => new Set(s).add("office")); }, []);

    99→  // ---- GUIDED DESIGN: apply or customize AI suggestions ----
   100→  const applySuggestion = useCallback((sg: GuidedSuggestion) => {
   101→    const catMap: Record<string, string> = { rug: "carpet", lighting: "lighting", plant: "plants", sofa: "furniture" };
   102→    const catSlug = catMap[sg.category] || "furniture";
   103→    setOpenCats((s) => new Set(s).add(catSlug));
   104→    const cat = CATEGORIES.find((c) => c.slug === catSlug);
   105→    if (cat && cat.subTypes[0]) {
   106→      setSelectedSubTypes((s) => ({ ...s, [catSlug]: [cat.subTypes[0].label] }));
   107→    }
   108→    setPrompt((p) => p ? `${p}\n${sg.title}` : sg.title);
   109→    toast(`پیشنهاد «${sg.title}» اعمال شد — برای ادامه طراحی بزن`);
   110→  }, [toast]);

   112→  const customizeSuggestion = useCallback((sg: GuidedSuggestion) => {
   113→    setPrompt(sg.desc);
   114→    toast("توضیحات پیشنهاد در فیلد دستور قرار گرفت — می‌تونی ویرایش کنی");
   115→  }, [toast]);
   116→
   117→  useEffect(() => {
   118→    if (presetSlug) { const p = products.find((x) => x.slug === presetSlug); if (p) { setPresetProduct(p); setSelected((s) => ({ ...s, [p.id]: p })); setTab("inspiration"); } } // eslint-disable-line react-hooks/set-state-in-effect
   119→  }, [presetSlug]);
   120→
   121→  // CONTINUE DESIGN: ?session=<id> restores a saved design session (uploaded
   122→  // room photo + prompt + style) instead of silently opening an empty studio.
   123→  // Continuation restore — same single-line disable pattern as presetSlug above.
   124→  useEffect(() => {
   125→    const sid = sp.get("session");
   126→    if (!sid || imageBase64) return;
   127→    const session = useDesignSessions.getState().sessions.find((s) => s.id === sid);
   128→    if (!session) return;
   129→    if (session.beforeImage) { setImageBase64(session.beforeImage); rs.loadRoom(session.beforeImage, style === "office" ? "فضای اداری" : "پذیرایی"); } // eslint-disable-line react-hooks/set-state-in-effect
   130→    if (session.prompt) setPrompt(session.prompt);
   131→    if (session.style) setStyle(session.style);
   132→    toast("طراحی ذخیره‌شده بازیابی شد — از همین‌جا ادامه بده");
   133→  }, [sp]); // eslint-disable-line react-hooks/exhaustive-deps
   134→
   135→  const handleFile = useCallback((file: File, set: (v: string) => void = setImageBase64) => {
   136→    if (!ALLOWED_TYPES.includes(file.type)) return toast("فقط JPG، PNG یا WEBP", "error");
   137→    if (file.size > MAX_FILE_SIZE) return toast("حداکثر حجم تصویر ۱۰ مگابایت", "error");

   139→    r.onerror = () => toast("خطا در خواندن فایل", "error");
   140→    r.onload = () => { const d = r.result as string; if (typeof d !== "string" || d.length < 100) return toast("فایل نامعتبر", "error"); set(d); if (set === setImageBase64) { rs.loadRoom(d, style === "office" ? "فضای اداری" : "پذیرایی"); analyzeRoom(d); } };

   149→      const pool = CAT_PRODUCTS[e.catSlug] || []; if (!pool.length) return; const i = cursor[e.catSlug] = (cursor[e.catSlug] ?? 0); cursor[e.catSlug]++; const prod = getProductById(pool[i % pool.length]); if (prod && !out.find((x) => x.id === prod.id)) out.push(prod);
   150→    });
   151→    return out;
   152→  }, [designElements]);

   170→      setSkuWarning(conflict.message || "توجه: دسته‌بندی انتخاب‌شده با کد محصول همخوانی ندارد.");
   171→    } else {
   172→      setSkuWarning(null);
   173→    }
   174→  }, [designElements]);
   175→
   176→  const generate = useCallback(async () => {
   177→    if (!imageBase64) return toast("عکس خانه را آپلود کن", "error");
   178→
   179→    const chosen: Product[] = placedProducts.length
   180→      ? placedProducts
   181→      : DEFAULT_ROOM_IDS.map(getProductById).filter(Boolean) as Product[];
   182→
   183→    const isFullSet = designElements.length === 0;
   184→    const uiIntent = detectIntent(prompt, style);
   185→    const uiScope = computeChangeScope(uiIntent);
   186→    const isFullRoom = isFullSet || uiIntent.type === "full_redesign";
   187→    const opCost = costOf(isFullRoom ? "generate" : (presetProduct ? "placement" : "edit"));
   188→
   189→    const lastTargets: RoomElement[] = (rs.placements ?? [])
   190→      .slice(-1)
   191→      .flatMap((p): RoomElement[] => {
   192→        const cat = (getProductById(p.productId)?.categorySlug ?? "").toLowerCase();
   193→        if (/furniture|sofa/.test(cat)) return ["sofa"];
   194→        if (/rug|carpet/.test(cat)) return ["rug"];
   195→        if (/curtain|textile/.test(cat)) return ["curtain"];
   196→        if (/light|lamp/.test(cat)) return ["lighting"];
   197→        if (/bed/.test(cat)) return ["bed"];
   198→        if (/tv/.test(cat)) return ["tv"];
   199→        if (/plant/.test(cat)) return ["plant"];
   200→        if (/art|decor/.test(cat)) return ["art"];
   201→        if (/shelf|bookcase/.test(cat)) return ["shelf"];
   202→        if (/table|chair|dining|office/.test(cat)) return ["table"];
   203→        return [];
   204→      });
   205→
   206→    const previousTargets: RoomElement[] = lastTargets.length ? lastTargets : uiScope.targets;
   207→    const previousChanges: string[] = rs.appliedChanges.slice(-3);
   208→
   209→    const pipelineProducts: PlacementProduct[] = chosen.map((p) => ({
   210→      id: p.id,
   211→      name: p.name,
   212→      category: p.categorySlug,
   213→      material: p.materials?.[0],
   214→      color: p.colors?.[0]?.name,
   215→      style: p.styleSlugs?.[0],
   216→      dimensions: parseProductDimensions(p.dimensions),
   217→    }));
   218→
   219→    const budgetNum = Number(budget.replace(/[^\d]/g, "")) || undefined;
   220→
   221→    const resolvedSkuProduct = skuInput.trim() ? getProductBySkuOrCode(skuInput.trim()) : undefined;
   222→    const categoryTargets = designElements.map((e) => categoryToRoomElement(e.catSlug, e.label));
   223→    const derivedTargets: RoomElement[] = deriveTargetsFromProducts(chosen);
   224→    const finalTargets: RoomElement[] = [...new Set([...uiScope.targets, ...derivedTargets, ...categoryTargets])];
   225→
   226→    const pipelineInput: PipelineInput = {
   227→      prompt: prompt || (isFullRoom ? "بازطراحی کامل فضا" : chosen.map((p) => p.name).join("، ")),
   228→      style,
   229→      room: style === "office" ? "فضای اداری" : (rs.roomType || "نشیمن"),
   230→      colors: roomAnalysis?.palette ?? rs.detectedColors ?? undefined,
   231→      scope: isFullRoom ? "full" : "targeted",
   232→      targets: finalTargets.length ? finalTargets : undefined,
   233→      preservedExtra: undefined,
   234→      referenceImage: imageBase64,
   235→      mask: undefined,
   236→      previousTargets: previousTargets.length ? previousTargets : undefined,
   237→      previousChanges: previousChanges.length ? previousChanges : undefined,
   238→      previousProductId: rs.placements?.[rs.placements.length - 1]?.productId,
   239→      previousSKU: getProductById(rs.placements?.[rs.placements.length - 1]?.productId ?? "")?.sku,
   240→      sku: skuInput.trim() || undefined,
   241→      productCode: skuInput.trim() || undefined,
   242→      productId: resolvedSkuProduct?.id ?? presetProduct?.id ?? (chosen.length === 1 ? chosen[0].id : undefined),
   243→      selection: designElements.length > 0
   244→        ? {
   245→            category: designElements[0].catSlug,
   246→            subTypes: designElements.map((e) => e.label),
   247→            targets: finalTargets,
   248→          }
   249→        : undefined,
   250→      products: pipelineProducts.length ? pipelineProducts : undefined,
   251→      budget: budgetNum ? { min: budgetNum, currency: "تومان" } : undefined,
   252→      roomUnderstanding: roomAnalysis
   253→        ? {
   254→            roomType: roomAnalysis.roomType,
   255→            layout: "unknown",
   256→            lighting: "unknown",
   257→            objects: [],
   258→            confidence: 0.6,
   259→          }
   260→        : undefined,
   261→    };
   262→
   263→    trackEvent("ai_started", {
   264→      metadata: { style: styleLabel, items: chosen.length, fullSet: isFullRoom, scope: uiScope.targets.join("، ") },
   265→    });
   266→
   267→    setLoading(true);
   268→    setError(null);
   269→    setPlacements([]);
   270→    setStage("ANALYZING_SPACE");
   271→
   272→    const stageTimer = setInterval(() => {
   273→      setStage((cur) => {
   274→        const i = STAGE_ORDER.indexOf(cur);
   275→        if (i < STAGE_ORDER.length - 1) return STAGE_ORDER[i + 1];
   276→        return cur;
   277→      });
   278→    }, 900);
   279→
   280→    const opLabel = isFullRoom ? "طراحی کامل" : (presetProduct ? "جای‌گذاری محصول" : "چیدمان وسایل");
   281→
   282→    const result = await useCredits.getState().runAiOperation(opLabel, opCost, async () => {
   283→      return aiService.pipeline(pipelineInput);
   284→    });
   285→
   286→    clearInterval(stageTimer);
   287→
   288→    if (!result.ok) {
   289→      setLoading(false);
   290→      if (result.reason === "insufficient") {
   291→        setError("اعتبار کافی نیست");
   292→        return toast("اعتبار کافی نیست", "error");
   293→      }
   294→      if (result.reason === "duplicate") {
   295→        setError("درخواست تکراری — نتیجه قبلی در حال پردازش است");
   296→        return;
   297→      }
   298→      setError("خطا در پردازش — لطفاً دوباره تلاش کن");
   299→      trackEvent("ai_failed", {});
   300→      return toast("خطا در پردازش AI — اعتبار برگردانده شد", "error");
   301→    }
   302→
   303→    try {
   304→      const pipelineRes: PipelineResult = result.result;
   305→      const inst = pipelineRes.instruction;
   306→      const plan: ProductPlacementPlan | undefined = pipelineRes.placement ?? inst.placement;
   307→
   308→      setLastIntent(uiIntent);
   309→      setLastScope({
   310→        targets: inst.targets.length ? inst.targets : uiScope.targets,
   311→        lockedElements: inst.preserved.length ? inst.preserved : uiScope.lockedElements,
   312→        summary: buildScopeSummary(pipelineRes, uiScope),
   313→      });
   314→
   315→      const renderedPlacements = plan && chosen.length >= 1
   316→        ? buildPlacementsFromPlan(chosen, plan)
   317→        : makePlacements(chosen);
   318→      setPlacements(renderedPlacements);
   319→
   320→      const editedImage = pipelineRes.result.afterImage;
   321→      const isPreview = !!pipelineRes.result.preview || editedImage === imageBase64;
   322→      const outputImage = isPreview ? imageBase64 : editedImage;
   323→
   324→      const realMatched = pipelineRes.matchedProducts && pipelineRes.matchedProducts.length > 0
   325→        ? pipelineRes.matchedProducts
   326→        : matchStoreProducts({
   327→            sku: skuInput.trim() || undefined,
   328→            productId: resolvedSkuProduct?.id,
   329→            targets: inst.targets.length ? inst.targets : finalTargets,
   330→            style,
   331→            roomType: rs.roomType,
   332→            budget: budgetNum,
   333→          });
   334→      setMatchedStoreProducts(realMatched);
   335→
   336→      rs.commitChange({
   337→        label: isFullRoom ? "چیدمان کامل" : (inst.targets.map((t) => t).join("، ") || uiScope.targets.join("، ")),
   338→        image: outputImage,
   339→        placements: renderedPlacements.map((pl) => ({
   340→          productId: pl.product.id,
   341→          category: pl.product.categorySlug,
   342→          reason: plan?.rationale ?? uiScope.summary,
   343→          placement: {
   344→            x: pl.xNorm,
   345→            y: pl.yNorm,
   346→            scale: pl.scale,
   347→            rotation: pl.rotation ?? 0,
   348→          },
   349→        })),
   350→        change: pipelineRes.intent.changes?.[0] ?? (prompt || uiScope.summary),
   351→        scope: pipelineRes.scope,
   352→      });
   353→
   354→      setStage("RENDERING");
   355→      trackEvent("ai_finished", {
   356→        metadata: {
   357→          count: chosen.length,
   358→          preview: isPreview,
   359→          engine: pipelineRes.imageEngine,
   360→          requestId: pipelineRes.requestId,
   361→        },
   362→      });
   363→
   364→      saveSession({
   365→        title: isFullRoom ? `طراحی ${styleLabel}` : (prompt || "چیدمان هوشمند"),
   366→        prompt: pipelineInput.prompt,
   367→        roomType: pipelineInput.room ?? "نشیمن",
   368→        style,
   369→        colors: pipelineInput.colors ?? [],
   370→        scope: isFullRoom ? "full" : "targeted",
   371→        targets: finalTargets,
   372→        status: isPreview ? "partial-success" : "success",
   373→        beforeImage: imageBase64,
   374→        afterImage: outputImage,
   375→        regions: [],
   376→        products: chosen.map((p) => ({ label: p.name, productId: p.id })),
   377→        creditsUsed: opCost,
   378→        preview: isPreview,
   379→        imageEngine: pipelineRes.imageEngine,
   380→      });
   381→
   382→      if (isPreview) {
   383→        toast("نتیجه به‌صورت پیش‌نمایش آماده شد (موتور تصویر در دسترس نیست)");
   384→      } else {
   385→        toast("چیدمان آماده شد");
   386→      }
   387→    } catch {
   388→      setError("خطا در نمایش نتیجه");
   389→    } finally {
   390→      setLoading(false);
   391→    }
   392→  }, [imageBase64, placedProducts, designElements, presetProduct, prompt, style, budget, skuInput, rs, roomAnalysis, styleLabel, toast, saveSession]);

   394→  const updatePlacement = useCallback((id: string, patch: Partial<Placement>) => setPlacements((ps) => ps.map((pl) => (pl.product.id === id ? { ...pl, ...patch } : pl))), []);
   395→  const removePlacement = useCallback((id: string) => setPlacements((ps) => ps.filter((pl) => pl.product.id !== id)), []);
   396→  const overlayCart = useCallback((p: Product) => { addToCart(p.id); toast("به سبد اضافه شد"); }, [addToCart, toast]);
   397→  const overlayWishlist = useCallback((p: Product) => { wl.toggleProduct(p.id); toast("به علاقه‌مندی اضافه شد"); }, [wl, toast]);
   398→  const overlayView = useCallback((p: Product) => router.push(`/products/${p.slug}`), [router]);

   400→  const placePresetInRoom = useCallback(async () => {
   401→    if (!imageBase64) return toast("عکس خانه را آپلود کن", "error");
   402→    if (!presetProduct) return;
   403→
   404→    const opCost = costOf("placement");
   405→    const pipelineProduct: PlacementProduct = {
   406→      id: presetProduct.id,
   407→      name: presetProduct.name,
   408→      category: presetProduct.categorySlug,
   409→      material: presetProduct.materials?.[0],
   410→      color: presetProduct.colors?.[0]?.name,
   411→      style: presetProduct.styleSlugs?.[0],
   412→      dimensions: parseProductDimensions(presetProduct.dimensions),
   413→    };
   414→
   415→    const targets: RoomElement[] = deriveTargetsFromProducts([presetProduct]);
   416→
   417→    const pipelineInput: PipelineInput = {
   418→      prompt: prompt || `محصول «${presetProduct.name}» را در اتاق قرار بده`,
   419→      style,
   420→      room: style === "office" ? "فضای اداری" : (rs.roomType || "نشیمن"),
   421→      scope: "targeted",
   422→      targets,
   423→      referenceImage: imageBase64,
   424→      productId: presetProduct.id,
   425→      products: [pipelineProduct],
   426→      previousTargets: deriveTargetsFromProducts(placedProducts),
   427→      previousChanges: rs.appliedChanges.slice(-3),
   428→    };
   429→
   430→    setLoading(true);
   431→    setError(null);
   432→    setPlacements([]);
   433→    setStage("ANALYZING_SPACE");
   434→
   435→    const timer = setInterval(() => {
   436→      setStage((cur) => {
   437→        const i = STAGE_ORDER.indexOf(cur);
   438→        if (i < STAGE_ORDER.length - 1) return STAGE_ORDER[i + 1];
   439→        return cur;
   440→      });
   441→    }, 900);
   442→
   443→    const result = await useCredits.getState().runAiOperation("جای‌گذاری محصول", opCost, async () => {
   444→      return aiService.pipeline(pipelineInput);
   445→    });
   446→
   447→    clearInterval(timer);
   448→
   449→    if (!result.ok) {
   450→      setLoading(false);
   451→      if (result.reason === "insufficient") {
   452→        setError("اعتبار کافی نیست");
   453→        return toast("اعتبار کافی نیست", "error");
   454→      }
   455→      if (result.reason === "duplicate") {
   456→        setError("درخواست تکراری — نتیجه قبلی در حال پردازش است");
   457→        return;
   458→      }
   459→      setError("خطا در پردازش");
   460→      return toast("خطا در جای‌گذاری — اعتبار برگردانده شد", "error");
   461→    }
   462→
   463→    try {
   464→      const pipelineRes: PipelineResult = result.result;
   465→      const plan = pipelineRes.placement ?? pipelineRes.instruction.placement;
   466→      const fallbackPlacement: Placement = { product: presetProduct, xNorm: 0.5, yNorm: 0.6, scale: 1 };
   467→      const newPlacements = plan
   468→        ? buildPlacementsFromPlan([presetProduct], plan)
   469→        : [fallbackPlacement];
   470→
   471→      const editedImage = pipelineRes.result.afterImage;
   472→      const isPreview = !!pipelineRes.result.preview || editedImage === imageBase64;
   473→      const outputImage = isPreview ? imageBase64 : editedImage;
   474→
   475→      setPlacements(newPlacements);
   476→      setStage("RENDERING");
   477→      setTab("design");
   478→
   479→      const matchedStore = pipelineRes.matchedProducts && pipelineRes.matchedProducts.length > 0
   480→        ? pipelineRes.matchedProducts
   481→        : matchStoreProducts({
   482→            productId: presetProduct.id,
   483→            targets,
   484→            style,
   485→            roomType: rs.roomType,
   486→          });
   487→      setMatchedStoreProducts(matchedStore);
   488→
   489→      rs.commitChange({
   490→        label: `جای‌گذاری ${presetProduct.name}`,
   491→        image: outputImage,
   492→        placements: newPlacements.map((pl) => ({
   493→          productId: pl.product.id,
   494→          category: pl.product.categorySlug,
   495→          reason: plan?.rationale ?? "جای‌گذاری محصول",
   496→          placement: { x: pl.xNorm, y: pl.yNorm, scale: pl.scale, rotation: pl.rotation ?? 0 },
   497→        })),
   498→        change: prompt || `قرار دادن ${presetProduct.name} در اتاق`,
   499→        scope: pipelineRes.scope,
   500→      });
   501→
   502→      saveSession({
   503→        title: `جای‌گذاری ${presetProduct.name}`,
   504→        prompt: pipelineInput.prompt,
   505→        roomType: pipelineInput.room ?? "نشیمن",
   506→        style,
   507→        colors: [],
   508→        scope: "targeted",
   509→        targets,
   510→        status: isPreview ? "partial-success" : "success",
   511→        beforeImage: imageBase64,
   512→        afterImage: outputImage,
   513→        regions: [],
   514→        products: [{ label: presetProduct.name, productId: presetProduct.id }],
   515→        creditsUsed: opCost,
   516→        preview: isPreview,
   517→        imageEngine: pipelineRes.imageEngine,
   518→      });
   519→
   520→      toast(isPreview ? "محصول به‌صورت پیش‌نمایش قرار گرفت (موتور تصویر در دسترس نیست)" : "محصول در عکس قرار گرفت");
   521→    } catch {
   522→      setError("خطا در نمایش نتیجه");
   523→    } finally {
   524→      setLoading(false);
   525→    }
   526→  }, [imageBase64, presetProduct, prompt, style, rs, placedProducts, toast, saveSession]);
   527→
   528→  const buyTheLook = useCallback(() => { if (!placedProducts.length) return; placedProducts.forEach((p) => addToCart(p.id)); toast("چیدمان به سبد اضافه شد"); }, [placedProducts, addToCart, toast]);
   529→  const handleSaveToWishlist = useCallback(() => {
   530→    if (!placedProducts.length) return toast("اول محصولاتی در طرح قرار بده", "info");
   531→    // Idempotent add — the old toggle could silently REMOVE everything on a
   532→    // second click while the toast kept claiming it was saved.
   533→    let added = 0;
   534→    placedProducts.forEach((p) => { if (!wl.products.includes(p.id)) { wl.toggleProduct(p.id); added += 1; } });
   535→    toast(added > 0 ? `${added} محصول طرح به علاقه‌مندی‌ها اضافه شد` : "همه محصولات این طرح قبلاً ذخیره شده‌اند");
   536→  }, [placedProducts, wl, toast]);
   537→
   538→  /** Clear the uploaded photo + every derived result (verbatim remove handler). */
   539→  const removeImage = useCallback(() => {
   540→    setImageBase64(null);
   541→    setPlacements([]);
   542→    rs.reset();
   543→    setRoomAnalysis(null);
   544→    setMatchedStoreProducts([]);
   545→  }, [rs]);
   546→
   547→  return {
   548→    // routing / tabs
   549→    router,
   550→    tab,
   551→    setTab,
   552→    // upload
   553→    imageBase64,
   554→    setImageBase64,
   555→    analyzing,
   556→    presetProduct,
   557→    setPresetProduct,
   558→    inspirationMatches,
   559→    setInspirationMatches,
   560→    handleFile,
   561→    // style / prompt / budget
   562→    style,
   563→    styleLabel,
   564→    selectStyle,
   565→    prompt,
   566→    setPrompt,
   567→    budget,
   568→    setBudget,
   569→    // items / sku
   570→    openCats,
   571→    toggleCat,
   572→    selectedSubTypes,
   573→    setSelectedSubTypes,
   574→    toggleSubType,
   575→    selected,
   576→    toggleProduct,
   577→    skuInput,
   578→    skuWarning,
   579→    handleSkuChange,
   580→    // analysis
   581→    roomAnalysis,
   582→    analyzeRoom,
   583→    applySuggestion,
   584→    customizeSuggestion,
   585→    // derived cart of chosen items
   586→    designElements,
   587→    placedProducts,
   588→    total,
   589→    // generation
   590→    cost,
   591→    loading,
   592→    stage,
   593→    error,
   594→    generate,
   595→    // result canvas
   596→    placements,
   597→    setPlacements,
   598→    removeImage,
   599→    updatePlacement,
   600→    removePlacement,
   601→    overlayCart,
   602→    overlayWishlist,
   603→    overlayView,
   604→    lastScope,
   605→    matchedStoreProducts,
   606→    addToCart,
   607→    // actions
   608→    buyTheLook,
   609→    handleSaveToWishlist,
   610→    placePresetInRoom,
   611→    // room-state (undo/redo + overlay mode)
   612→    rs,
   613→    wl,
   614→    toast,
   615→    saveSession,
   616→  };
   617→}
