   446→export const productsByStore = (storeId: string) =>

   451→// ============================================================
   452→// DISCOVERY — Similar products (style + category overlap).
   453→// ============================================================
   454→export const similarProducts = (productId: string, limit = 4): Product[] => {
   455→  const target = products.find((p) => p.id === productId);
   456→  if (!target) return [];
   457→  return products
   458→    .filter((p) => p.id !== productId)
   459→    .map((p) => {
   460→      const sharedStyles = p.styleSlugs.filter((s) => target.styleSlugs.includes(s)).length;
   461→      const sameCategory = p.categorySlug === target.categorySlug ? 1 : 0;
   462→      const sameStore = p.storeId === target.storeId ? 1 : 0;
   463→      return { p, score: sharedStyles * 2 + sameCategory + sameStore };
   464→    })
   465→    .filter((x) => x.score > 0)
   466→    .sort((a, b) => b.score - a.score)
   467→    .slice(0, limit)
   468→    .map((x) => x.p);
   469→};

   474→// flickers between renders. Replace with real sales from backend.

   477→  Math.max(1, Math.round(p.reviewsCount * 6 + p.rating * 12));

   496→export const getProductBySku = (sku: string): Product | undefined => {
   497→  if (!sku) return undefined;
   498→  const clean = sku.trim().toLowerCase();
   499→  if (SKU_ALIASES[clean]) {
   500→    const aliased = getProductById(SKU_ALIASES[clean]);
   501→    if (aliased) return aliased;
   502→  }
   503→  return products.find(
   504→    (p) =>
   505→      (p.sku && p.sku.toLowerCase() === clean) ||
   506→      p.id.toLowerCase() === clean ||
   507→      p.slug.toLowerCase() === clean
   508→  );
   509→};
