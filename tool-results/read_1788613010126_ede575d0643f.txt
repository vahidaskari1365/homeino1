   145→   145→  const placedProducts = useMemo(() => {

   149→   149→      const pool = CAT_PRODUCTS[e.catSlug] || []; if (!pool.length) return; const i = cursor[e.catSlug] = (cursor[e.catSlug] ?? 0); cursor[e.catSlug]++; const prod = getProductById(pool[i % pool.length]); if (prod && !out.find((x) => x.id === prod.id)) out.push(prod);
   150→   150→    });

   188→   188→
   189→   189→    const lastTargets: RoomElement[] = (rs.placements ?? [])

   197→   197→        if (/bed/.test(cat)) return ["bed"];
   198→   198→        if (/tv/.test(cat)) return ["tv"];
   199→   199→        if (/plant/.test(cat)) return ["plant"];
   200→   200→        if (/art|decor/.test(cat)) return ["art"];
   201→   201→        if (/shelf|bookcase/.test(cat)) return ["shelf"];
   202→   202→        if (/table|chair|dining|office/.test(cat)) return ["table"];

   320→   320→      const editedImage = pipelineRes.result.afterImage;

   324→   324→      const realMatched = pipelineRes.matchedProducts && pipelineRes.matchedProducts.length > 0

   328→   328→            productId: resolvedSkuProduct?.id,

   331→   331→            roomType: rs.roomType,
   332→   332→            budget: budgetNum,
   333→   333→          });

   335→   335→
   336→   336→      rs.commitChange({
   337→   337→        label: isFullRoom ? "چیدمان کامل" : (inst.targets.map((t) => t).join("، ") || uiScope.targets.join("، ")),
   338→   338→        image: outputImage,
   339→   339→        placements: renderedPlacements.map((pl) => ({
   340→   340→          productId: pl.product.id,
   341→   341→          category: pl.product.categorySlug,
   342→   342→          reason: plan?.rationale ?? uiScope.summary,
   343→   343→          placement: {
   344→   344→            x: pl.xNorm,
   345→   345→            y: pl.yNorm,
   346→   346→            scale: pl.scale,
   347→   347→            rotation: pl.rotation ?? 0,
   348→   348→          },
   349→   349→        })),
   350→   350→        change: pipelineRes.intent.changes?.[0] ?? (prompt || uiScope.summary),
   351→   351→        scope: pipelineRes.scope,
   352→   352→      });
   353→   353→
   354→   354→      setStage("RENDERING");
   355→   355→      trackEvent("ai_finished", {
   356→   356→        metadata: {
   357→   357→          count: chosen.length,
   358→   358→          preview: isPreview,
   359→   359→          engine: pipelineRes.imageEngine,
   360→   360→          requestId: pipelineRes.requestId,

   430→   430→    setLoading(true);

   502→   502→      saveSession({

   514→   514→        products: [{ label: presetProduct.name, productId: presetProduct.id }],

   521→   521→    } catch {
   522→   522→      setError("خطا در نمایش نتیجه");
   523→   523→    } finally {
